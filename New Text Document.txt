<script>
function buatHati() {
  const hati = document.createElement("div");
  hati.className = "terapung";

  const emoji = ["❤️", "💗", "💕", "💖", "💘", "💝"];
  hati.textContent =
    emoji[Math.floor(Math.random() * emoji.length)];

  hati.style.left = Math.random() * 100 + "vw";
  hati.style.fontSize = (18 + Math.random() * 25) + "px";
  hati.style.animationDuration = (4 + Math.random() * 4) + "s";

  document.body.appendChild(hati);

  setTimeout(() => hati.remove(), 8000);
}

setInterval(buatHati, 650);

function bukaSurprise() {
  document.getElementById("mesej").style.display = "block";

  const butang = document.querySelector(".card > button");
  butang.style.display = "none";

  for (let i = 0; i < 15; i++) {
    setTimeout(buatHati, i * 100);
  }
}

function hatiBanyak() {
  for (let i = 0; i < 35; i++) {
    setTimeout(buatHati, i * 100);
  }
}
